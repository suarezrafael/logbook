\begin{definition}
For output $g_e$, target $b$, and one support variable $v$, the fiber
$g_e^{-1}(b)$ is \emph{functional in head $v$} when projection onto the other
support variables determines $v$. Equivalently the fiber is contained in a
partial graph $x_v=h(\text{tails})$. Extend $h$ arbitrarily to a total Boolean
function.
\end{definition}

\begin{lemma}[Safe graph relaxation]
Replacing a selected functional fiber by its total graph relation and deleting
that output coordinate is safe for range avoidance: every missing word on the
unselected outputs over the relaxed graph domain lifts by inserting the selected
target bit.
\end{lemma}
\begin{proof}
Every original realization of the selected output belongs to the fiber and
hence to the graph; total extension only enlarges the allowed domain.
\end{proof}

\begin{theorem}[Functional-DAG avoider]
Select $s$ functional anchors with distinct head variables such that all tail to
head dependencies form a DAG. Let $\mu=n-s$ be the number of non-head variables.
Then a missing output is constructible in
\[
O(2^\mu\operatorname{poly}(N)).
\]
\end{theorem}
\begin{proof}
Every assignment to the $\mu$ roots extends uniquely in topological order to a
full assignment satisfying all relaxed functional graphs. Thus the relaxed
domain has exactly $2^\mu$ points. After deleting the $s$ selected outputs,
$m-s>n-s=\mu$ output coordinates remain, so their map cannot cover its full
output cube. Enumerate the roots, choose a missing remaining word, and lift.
\end{proof}

\begin{lemma}
Every unbalanced essential ternary predicate has a functional fiber.
\end{lemma}
\begin{proof}
A smaller fiber has at most three cube vertices. Failure of functionality in a
coordinate means the fiber contains a cube edge in that direction. At most
three cube vertices cannot contain edges in all three directions because the
Boolean cube is bipartite and has no triangle. Hence some coordinate projection
is injective on the fiber.
\end{proof}

\begin{lemma}
An essential balanced ternary predicate with a functional fiber is either affine
or NPN-equivalent to the graph-of-AND orbit $0x1e$.
\end{lemma}
\begin{proof}
A balanced functional fiber has four points and is the graph
$z=h(x,y)$ of a total two-input function. Essentiality forces $h$ to depend on
both inputs. If $h$ is affine the ternary graph predicate is affine; otherwise
all essential non-affine two-input Boolean functions are NPN-equivalent to AND,
giving orbit $0x1e$.
\end{proof}

\begin{theorem}[Exact ternary boundary]
Exactly $186$ of the $218$ essential ternary truth tables have a functional
anchor. The only anchor-free NPN orbits are signed majority $0x17$ (eight
tables) and mux/bijunctive $0x1b$ (twenty-four tables).
\end{theorem}

\begin{theorem}[Strict balanced-nonaffine family]
For every $N\ge5$, put canonical $0x1e$ gates on all cyclic triples and add one
majority output. V97 has $\lambda=N$, V100 performs no peel, while V101 has
$\mu=2$ and enumerates four relaxed assignments.
\end{theorem}
\begin{proof}
Canonical $0x1e$ satisfies
\[
f(x_0,x_1,x_2)=x_2\oplus(x_0\lor x_1).
\]
For target zero, cyclic output $i$ gives
$x_{i+2}=x_i\lor x_{i+1}$. Select outputs $0,\ldots,N-3$; their distinct heads
$x_2,\ldots,x_{N-1}$ form an acyclic recurrence rooted at $x_0,x_1$. The support
is connected with minimum input degree at least three, and both $0x1e$ and
majority lie outside V100's peelable orbits.
\end{proof}

\paragraph{Boundary.}
High-root systems and NPN orbits $0x17/0x1b$ remain open. No unrestricted
$NC^0_3$, novelty, lower-bound, or P-versus-NP claim is made.
